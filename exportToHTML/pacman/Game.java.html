// Simple PacMan implementation
package pacman;

import ch.aplu.jgamegrid.*;
import pacman.utility.GameCallback;

import java.awt.*;
import java.util.List;
import java.util.Properties;

public class Game extends GameGrid
{
  private final static int DEFAULT_HORZ_CELLS = 20;
  private final static int DEFAULT_VERT_CELLS = 11;
  private final static int DEFAULT_SIMULATION_PERIOD_MS = 300;
  private final int nbHorzCells;
  private final int nbVertCells;
  private GameController gameController;
  private GameCallback gameCallback;

  public Game(GameCallback gameCallback, Properties properties)
  {
    //Setup game
    super(DEFAULT_HORZ_CELLS, DEFAULT_VERT_CELLS, 20, false);
    this.nbHorzCells = DEFAULT_HORZ_CELLS;
    this.nbVertCells = DEFAULT_VERT_CELLS;
    this.gameCallback = gameCallback;
    gameController = new GameController(nbHorzCells, nbVertCells, properties, gameCallback);
    setSimulationPeriod(resolveSimulationPeriod(properties));
    setTitle("[PacMan in the Multiverse]");

    GGBackground bg = getBg();
    drawGrid(bg);

    addKeyRepeatListener(gameController.getPacActor().getPacmanController());
    setKeyRepeatPeriod(150);
    setupActorLocations();
  }

  public GameCallback getGameCallback() {
    return gameCallback;
  }

  /**
   * GameGrid's own act() fires once per simulation frame - a single,
   * well-defined place to advance the tick counter that every actor's log
   * line reads from, rather than each actor incrementing (and racing on)
   * its own. Every actor now acts on every frame (no per-actor
   * setSlowDown()), so one tick = one move for PacMan and every monster;
   * pacing is controlled purely via the simulation period instead.
   */
  @Override
  public void act() {
    super.act();
    gameController.incrementTick();
    updateTitle();
  }

  private void updateTitle() {
    setTitle("[PacMan in the Multiverse] Tick: " + gameController.getCurrentTick()
            + " Score: " + gameController.getPacActor().getScore());
  }

  /**
   * How many milliseconds elapse per simulation frame - i.e. per tick,
   * and (since every actor now acts every frame) per move. JGameGrid's
   * own setSimulationPeriod(). Configurable via the {@code
   * Simulation.period} property so automated tests can run the same
   * deterministic move scripts much faster than real gameplay speed
   * without changing tick counts, move logic, or any other behaviour -
   * only wall-clock speed changes. Defaults to 300ms when absent, which
   * matches the original per-move pacing from when PacMan/monsters used
   * setSlowDown(3) at a 100ms simulation period (3 x 100ms = 300ms).
   */
  private static int resolveSimulationPeriod(Properties properties) {
    String period = properties.getProperty("Simulation.period");
    if (period == null || period.isEmpty()) {
      return DEFAULT_SIMULATION_PERIOD_MS;
    }
    return Integer.parseInt(period.trim());
  }

  /**
   * Run the game to completion (win/loss) and return the full event log,
   * so this can be driven by a properties-in / logResult-out JUnit test.
   */
  public String runApp() {
    //Run the game
    doRun();
    show();
    gameController.setupPillAndItemsLocations();

    do {
      gameController.checkEndGame();

      delay(10);
    } while(!gameController.hasEndGame());
    delay(120);

    Location loc = gameController.getPacActor().getLocation();
    gameController.stopMonsters();
    gameController.getPacActor().removeSelf();

    String title = "";
    if (gameController.isHasPacmanBeenHit()) {
      getBg().setPaintColor(Color.red);
      title = "GAME OVER";
      addActor(new Actor("sprites/explosion3.gif"), loc);
    } else if (gameController.isHasPacmanEatAllPills()) {
      getBg().setPaintColor(Color.yellow);
      title = "YOU WIN";
    }
    setTitle(title);
    gameCallback.endOfGame(title);

    doPause();

    return gameCallback.getAllLog();
  }

  private void setupActorLocations() {
    List<ActorLocation> actorLocations = gameController.getActorLocations();
    for (ActorLocation actorLocation: actorLocations) {
      addActor(actorLocation.getActor(), actorLocation.getLocation(), actorLocation.getDirection());
    }
  }

  private void drawGrid(GGBackground bg)
  {
    bg.clear(Color.gray);
    bg.setPaintColor(Color.white);
    for (int y = 0; y < nbVertCells; y++)
    {
      for (int x = 0; x < nbHorzCells; x++)
      {
        bg.setPaintColor(Color.white);
        Location location = new Location(x, y);
        boolean isNormal = gameController.isNormal(location);
        boolean isPill = gameController.isPill(location);
        boolean isGold = gameController.isGold(location);
        boolean isIce = gameController.isIce(location);
        if (isNormal)
          bg.fillCell(location, Color.lightGray);
        if (isPill)
          putPill(bg, location);
        if (isGold)
          putGold(bg, location);
        if (isIce)
          putIce(bg, location);
      }
    }

    for (Location location : gameController.getPropertyPillLocations()) {
      putPill(bg, location);
    }

    for (Location location : gameController.getPropertyGoldLocations()) {
      putGold(bg, location);
    }
  }

  private void putPill(GGBackground bg, Location location){
    bg.fillCircle(toPoint(location), 5);
  }

  private void putGold(GGBackground bg, Location location){
    bg.setPaintColor(Color.yellow);
    bg.fillCircle(toPoint(location), 5);
    Actor gold = new Actor("sprites/gold.png");
    gameController.putGold(gold);
    addActor(gold, location);
  }

  private void putIce(GGBackground bg, Location location){
    bg.setPaintColor(Color.blue);
    bg.fillCircle(toPoint(location), 5);
    Actor ice = new Actor("sprites/ice.png");
    gameController.putIce(ice);
    addActor(ice, location);
  }

}
